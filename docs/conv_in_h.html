study_parameters <- list(problem = convergence_parameters, alpha = alpha_vector,
                         h = h_vector, tau = time_step_fixed, m = m_fixed)

For these smooth finite-mode data, the spatial benchmark is second order. We vary h over two orders of magnitude, keeping tau=1e-10 and m=8 fixed. The helper evaluates the discrete FEM/backward Euler recurrence in closed form; it does not allocate or step through the ten-billion-interval grid.

# Compute the error matrix and save at the original main-branch location.
errors_projected <- matrix(NA_real_, length(h_vector), length(alpha_vector))
for (i in seq_along(h_vector)) {
  for (j in seq_along(alpha_vector)) {
    errors_projected[i, j] <- tadpole_convergence_error(
      h = h_vector[i], time_step = time_step_fixed, m = m_fixed,
      alpha = alpha_vector[j], parameters = convergence_parameters
    )
  }
}
save(errors_projected, study_parameters,
     file = here::here("data_files/errors_projected_h.RData"))
# Verify that the fixed-name results belong to the parameters above.
saved <- new.env()
load(here::here("data_files/errors_projected_h.RData"), envir = saved)
stopifnot(identical(saved$study_parameters, study_parameters))
errors_projected <- saved$errors_projected
observed_rates <- numeric(length(alpha_vector))
for (u in 1:length(alpha_vector)) {
  observed_rates[u] <- coef(lm(log10(errors_projected[, u]) ~ log10(h_vector)))[2]
}

# Smooth finite-mode data give the saturated second-order benchmark.
theoretical_rates <- rep(2, length(alpha_vector))
