tau_vector <- vapply(study_range, function(tau)
  experiment_grid(convergence_parameters$T_final, tau)$time_step, numeric(1))
study_parameters <- list(problem = convergence_parameters, alpha = alpha_vector,
                         h = h_fixed, tau = tau_vector, m = m_fixed)

The backward Euler time benchmark is first order. We vary the actual uniform time step over two orders of magnitude, keeping h=1e-5 and m=8 fixed. The helper uses interval averages of the physical sinusoidal forcing and the same right-constant reconstruction as the direct time-stepping solver.

# Compute the error matrix and save at the original main-branch location.
errors_projected <- matrix(NA_real_, length(tau_vector), length(alpha_vector))
for (i in seq_along(tau_vector)) {
  for (j in seq_along(alpha_vector)) {
    errors_projected[i, j] <- tadpole_convergence_error(
      h = h_fixed, time_step = tau_vector[i], m = m_fixed,
      alpha = alpha_vector[j], parameters = convergence_parameters
    )
  }
}
save(errors_projected, study_parameters,
     file = here::here("data_files/errors_projected_tau.RData"))
# Verify that the fixed-name results belong to the parameters above.
saved <- new.env()
load(here::here("data_files/errors_projected_tau.RData"), envir = saved)
stopifnot(identical(saved$study_parameters, study_parameters))
errors_projected <- saved$errors_projected
observed_rates <- numeric(length(alpha_vector))
for (u in 1:length(alpha_vector)) {
  observed_rates[u] <- coef(lm(log10(errors_projected[, u]) ~ log10(tau_vector)))[2]
}

theoretical_rates <- rep(1, length(alpha_vector))
